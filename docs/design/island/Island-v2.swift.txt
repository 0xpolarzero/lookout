/// Warm neutral greys (Linear's 2026 refresh moved off blue-grey), one accent, one attention colour.
enum T {
    static let ink = Color.black                                   // the island: the notch's own black
    static let raised = Color(red: 0.085, green: 0.083, blue: 0.08)
    static let hover = Color(red: 0.115, green: 0.112, blue: 0.108)
    static let line = Color.white.opacity(0.065)
    static let text = Color(red: 0.93, green: 0.925, blue: 0.91)
    static let text2 = Color(red: 0.60, green: 0.595, blue: 0.58)
    static let text3 = Color(red: 0.40, green: 0.395, blue: 0.385)
    static let accent = Color(red: 0.45, green: 0.47, blue: 0.93)   // indigo, for done and the one primary action
    static let amber = Color(red: 0.93, green: 0.68, blue: 0.27)    // only for what needs you
    static let red = Color(red: 0.90, green: 0.38, blue: 0.36)
    static let green = Color(red: 0.36, green: 0.72, blue: 0.50)
    static let violet = Color(red: 0.62, green: 0.54, blue: 0.92)

    static func body(_ w: Font.Weight = .regular) -> Font { .system(size: 13, weight: w) }
    static func meta(_ w: Font.Weight = .regular) -> Font { .system(size: 12, weight: w) }
    static let micro = Font.system(size: 11, weight: .medium)
}

// MARK: - Status glyphs (Linear-style, drawn, 14pt)

enum Kind { case question, plan, done, stuck, working, quiet, failing, passing, running }

struct Glyph: View {
    let kind: Kind
    var size: CGFloat = 14
    var body: some View {
        let s = size, lw = max(1.2, s * 0.1)
        ZStack {
            switch kind {
            case .question:
                Circle().fill(T.amber)
                Text("?").font(.system(size: s * 0.66, weight: .heavy, design: .rounded)).foregroundStyle(.black).offset(y: -s * 0.02)
            case .plan:
                Circle().strokeBorder(T.violet, lineWidth: lw)
                Pie(fraction: 0.75).fill(T.violet).padding(s * 0.24)
            case .done:
                Circle().fill(T.accent)
                Image(systemName: "checkmark").font(.system(size: s * 0.48, weight: .black)).foregroundStyle(.black)
            case .stuck:
                Circle().fill(T.red)
                Text("!").font(.system(size: s * 0.66, weight: .heavy, design: .rounded)).foregroundStyle(.black)
            case .working:
                Circle().strokeBorder(T.text2, lineWidth: lw)
                Pie(fraction: 0.5).fill(T.text2).padding(s * 0.24)
            case .quiet:
                Circle().fill(T.text3)
                Image(systemName: "checkmark").font(.system(size: s * 0.48, weight: .black)).foregroundStyle(.black)
            case .failing:
                Circle().fill(T.red)
                Image(systemName: "xmark").font(.system(size: s * 0.44, weight: .black)).foregroundStyle(.black)
            case .passing:
                Circle().fill(T.green)
                Image(systemName: "checkmark").font(.system(size: s * 0.48, weight: .black)).foregroundStyle(.black)
            case .running:
                Circle().strokeBorder(T.text2, style: StrokeStyle(lineWidth: lw, dash: [s * 0.14, s * 0.12]))
                Pie(fraction: 0.33).fill(T.text2).padding(s * 0.24)
            }
        }
        .frame(width: s, height: s)
    }
}

struct Pie: Shape {
    var fraction: Double
    func path(in r: CGRect) -> Path {
        var p = Path()
        let c = CGPoint(x: r.midX, y: r.midY)
        p.move(to: c)
        p.addArc(center: c, radius: r.width / 2, startAngle: .degrees(-90), endAngle: .degrees(-90 + 360 * fraction), clockwise: false)
        p.closeSubpath()
        return p
    }
}

// MARK: - The island's shape

/// Flat on top, flared into the screen's edge like the notch, rounded below.
struct IslandShape: Shape {
    var radius: CGFloat
    var flare: CGFloat = 9
    func path(in r: CGRect) -> Path {
        let w = r.width, h = r.height, f = flare, R = min(radius, h / 2)
        var p = Path()
        p.move(to: .zero)
        p.addQuadCurve(to: CGPoint(x: f, y: f), control: CGPoint(x: f, y: 0))
        p.addLine(to: CGPoint(x: f, y: h - R))
        p.addQuadCurve(to: CGPoint(x: f + R, y: h), control: CGPoint(x: f, y: h))
        p.addLine(to: CGPoint(x: w - f - R, y: h))
        p.addQuadCurve(to: CGPoint(x: w - f, y: h - R), control: CGPoint(x: w - f, y: h))
        p.addLine(to: CGPoint(x: w - f, y: f))
        p.addQuadCurve(to: CGPoint(x: w, y: 0), control: CGPoint(x: w - f, y: 0))
        p.closeSubpath()
        return p
    }
}

struct Island<Content: View>: View {
    var width: CGFloat
    var radius: CGFloat = 22
    @ViewBuilder var content: Content
    var body: some View {
        content
            .frame(width: width - 18)
            .padding(.horizontal, 9)
            .background(IslandShape(radius: radius).fill(T.ink))
            .shadow(color: .black.opacity(0.35), radius: 30, y: 14)
    }
}

/// The row that sits beside the camera: what's left and right of the notch.
struct Ears<L: View, R: View>: View {
    @ViewBuilder var left: L
    @ViewBuilder var right: R
    var body: some View {
        HStack(spacing: 0) {
            left.frame(maxWidth: .infinity, alignment: .leading)
            Color.clear.frame(width: Notch.size.width)
            right.frame(maxWidth: .infinity, alignment: .trailing)
        }
        .frame(height: Notch.size.height)
        .padding(.horizontal, 14)
    }
}

enum Notch { static let size = CGSize(width: 192, height: 32) }

// MARK: - Screen (a 14" MacBook Pro, its top part)

struct Screen<Content: View>: View {

    var size = CGSize(width: 1512, height: 982)
    var wallpaper = "/System/Library/Desktop Pictures/Sonoma.heic"
    @ViewBuilder var content: Content

    var body: some View {
        ZStack(alignment: .top) {
            if let img = NSImage(contentsOfFile: wallpaper) {
                Image(nsImage: img).resizable().aspectRatio(contentMode: .fill).frame(width: size.width, height: size.height).clipped()
            }
            appBehind.offset(x: -210, y: 150)
            menuBar
            content
        }
        .frame(width: size.width, height: size.height, alignment: .top)
        .clipped()
    }

    var menuBar: some View {
        HStack(spacing: 20) {
            Image(systemName: "apple.logo").font(.system(size: 14))
            Text("Claude").font(.system(size: 13, weight: .bold))
            ForEach(["File", "Edit", "View", "Window", "Help"], id: \.self) { Text($0).font(.system(size: 13)) }
            Spacer()
            Image(systemName: "wifi").font(.system(size: 13))
            Image(systemName: "battery.75percent").font(.system(size: 15))
            Image(systemName: "switch.2").font(.system(size: 12))
            Text("Thu 9 Oct  14:32").font(.system(size: 13))
        }
        .foregroundStyle(.white.opacity(0.92))
        .shadow(color: .black.opacity(0.25), radius: 2)
        .padding(.horizontal, 18)
        .frame(height: Notch.size.height)
        .overlay {
            // The camera housing: Lookout's island grows out of it.
            IslandShape(radius: 10, flare: 0).fill(.black).frame(width: Notch.size.width, height: Notch.size.height)
        }
    }

    var appBehind: some View {
        VStack(alignment: .leading, spacing: 0) {
            HStack(spacing: 8) { ForEach(0..<3) { _ in Circle().fill(Color.white.opacity(0.18)).frame(width: 12, height: 12) } }
                .padding(14)
            ForEach([0.5, 0.82, 0.7, 0.4, 0.76, 0.6], id: \.self) { w in
                RoundedRectangle(cornerRadius: 4).fill(Color.white.opacity(0.06)).frame(width: 620 * w, height: 10).padding(.horizontal, 22).padding(.vertical, 9)
            }
            Spacer()
        }
        .frame(width: 760, height: 560, alignment: .topLeading)
        .background(RoundedRectangle(cornerRadius: 14).fill(Color(white: 0.13)))
        .overlay(RoundedRectangle(cornerRadius: 14).strokeBorder(Color.white.opacity(0.08)))
        .shadow(color: .black.opacity(0.3), radius: 30, y: 10)
    }
}

struct CardRow: View {
    let card: Card
    var selected = false
    var twoLines = true
    var body: some View {
        HStack(alignment: .top, spacing: 10) {
            Glyph(kind: card.kind).padding(.top, 1.5)
            VStack(alignment: .leading, spacing: 3) {
                HStack(spacing: 6) {
                    Text(card.title).font(T.body(.medium)).foregroundStyle(T.text).lineLimit(1)
                    Spacer(minLength: 8)
                    Text(card.ago).font(T.meta()).foregroundStyle(T.text3).monospacedDigit()
                }
                if twoLines {
                    HStack(spacing: 5) {
                        Text(card.project).foregroundStyle(T.text2)
                        Text("·").foregroundStyle(T.text3)
                        Text(card.line).foregroundStyle(T.text3).lineLimit(1)

                }
            }
        }
        .padding(.horizontal, 10).padding(.vertical, 8)
        .background(RoundedRectangle(cornerRadius: 9).fill(selected ? T.hover : .clear))
    }
}

enum Space: String, CaseIterable { case forYou = "For you", inbox = "Inbox", ci = "CI" }

struct SpaceBar: View {
    var current: Space
    var body: some View {
        HStack(spacing: 2) {
            ForEach(Space.allCases, id: \.self) { s in
                HStack(spacing: 6) {
                    Text(s.rawValue).font(T.meta(.medium)).foregroundStyle(s == current ? T.text : T.text3)
                    switch s {
                    case .forYou: Text("4").font(T.micro).foregroundStyle(s == current ? T.amber : T.text3)
                    case .inbox: Text("5").font(T.micro).foregroundStyle(T.text3)
                    case .ci: Circle().fill(T.red).frame(width: 5, height: 5)
                    }
                }
                .padding(.horizontal, 10).frame(height: 24)
                .background(Capsule().fill(s == current ? T.hover : .clear))
            }
        }
    }
}

struct Composer: View {
    var to: String?
    var text: String? = nil
    var body: some View {
        HStack(spacing: 8) {
            if let to {
                HStack(spacing: 5) { Mono(name: to); Text(to).font(T.meta(.medium)).foregroundStyle(T.text) }
                    .padding(.leading, 3).padding(.trailing, 8).frame(height: 22)
                    .background(RoundedRectangle(cornerRadius: 6).fill(T.hover))
            }
            if let text {
                (Text(text).foregroundStyle(T.text) + Text("|").foregroundStyle(T.accent)).font(T.body())
            } else {
                Text(to == nil ? "Tell the Router, or @ a session" : "Reply, or tell the Router").font(T.body()).foregroundStyle(T.text3)
            }
            Spacer()
            Key(k: "@")
            Key(k: "↵")

        .background(RoundedRectangle(cornerRadius: 11).fill(T.raised))
        .overlay(RoundedRectangle(cornerRadius: 11).strokeBorder(T.line))
    }
}

/// An entry in a card's thread: what happened, said small, as Linear's issue activity.

@main
struct Render {
    @MainActor static func main() {
        let dir = CommandLine.arguments.dropFirst().first ?? "shots"
        try? FileManager.default.createDirectory(atPath: dir, withIntermediateDirectories: true)
        let top = CGSize(width: 1512, height: 982)
        save(Rest(attention: false), "1a-rest-quiet", dir, crop: CGRect(x: 356, y: 0, width: 800, height: 220))
        save(Rest(attention: true), "1b-rest-needs-you", dir, crop: CGRect(x: 356, y: 0, width: 800, height: 220))
        save(Arrival(), "2-question-arrives", dir, crop: CGRect(x: 256, y: 0, width: 1000, height: 340))
        save(Glance(), "3-glance", dir, crop: CGRect(x: 256, y: 0, width: 1000, height: 480))
        save(RouterOpen(focus: .question), "4-router-question", dir, crop: CGRect(origin: .zero, size: top))
        save(RouterOpen(focus: .plan), "5-router-plan", dir, crop: CGRect(origin: .zero, size: top))
        save(InboxOpen(), "6-inbox", dir, crop: CGRect(x: 0, y: 0, width: 1512, height: 720))
        save(GlyphSheet(), "7-glyphs", dir, crop: nil)
    }

    @MainActor static func save<V: View>(_ view: V, _ name: String, _ dir: String, crop: CGRect?) {
        let r = ImageRenderer(content: view.environment(\.colorScheme, .dark))
        r.scale = 2
        guard var cg = r.cgImage else { print("failed \(name)"); return }
        if let crop, let c = cg.cropping(to: CGRect(x: crop.minX * 2, y: crop.minY * 2, width: crop.width * 2, height: crop.height * 2)) { cg = c }
        let rep = NSBitmapImageRep(cgImage: cg)
        try? rep.representation(using: .png, properties: [:])?.write(to: URL(fileURLWithPath: "\(dir)/\(name).png"))
        print("\(dir)/\(name).png")
    }
}
